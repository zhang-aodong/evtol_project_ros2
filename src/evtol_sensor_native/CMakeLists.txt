cmake_minimum_required(VERSION 3.8)
project(evtol_sensor_native)
find_package(ament_cmake REQUIRED)
add_library(evtol_sensor_native SHARED semantic.cpp)
target_compile_features(evtol_sensor_native PRIVATE cxx_std_11)
if(MSVC)
  target_compile_options(evtol_sensor_native PRIVATE /O2)
else()
  target_compile_options(evtol_sensor_native PRIVATE -O3)
endif()
install(TARGETS evtol_sensor_native RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
ament_package()
